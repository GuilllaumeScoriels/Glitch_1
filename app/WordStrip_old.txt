@OptIn(ExperimentalAnimationApi::class)
@Composable
fun WordStrip(prev: String,
              current: String,
              next: String,
              isFullScreen: Boolean,
              onExit: () -> Unit
) {
    if (isFullScreen) {
        AnimatedContent(
            targetState = Triple(prev, current, next), // on surveille les 3 mots ensemble
            transitionSpec = {
                // Entrée par la droite, sortie par la gauche
                slideInHorizontally { fullWidth -> fullWidth } with
                        slideOutHorizontally { fullWidth -> -fullWidth }
            }
        ) { (prevWord, currentWord, nextWord) ->
            Row(
                modifier = Modifier
                    .animateContentSize()
                    .fillMaxSize()
                    .background(Color.Black)
                    .clickable { onExit() },
                verticalAlignment = Alignment.CenterVertically
            ) {
                // Zone gauche : occupe la moitié de l'espace restant
                Box(modifier = Modifier.weight(1f)) {
                    Text(
                        text = prev,
                        color = Color.LightGray,
                        fontSize = 28.sp,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis,
                    )
                }

                Spacer(modifier = Modifier.width(12.dp))

                // Mot central : reste parfaitement centré
                AnimatedWord(current)


                Spacer(modifier = Modifier.width(12.dp))

                // Zone droite : occupe l’autre moitié de l'espace restant
                Box(modifier = Modifier.weight(1f)) {
                    Text(
                        text = next,
                        color = Color.LightGray,
                        fontSize = 28.sp,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis,
                    )
                }
            }
        }
    }
    else
        AnimatedContent(
            targetState = Triple(prev, current, next), // on surveille les 3 mots ensemble
            transitionSpec = {
                // Entrée par la droite, sortie par la gauche
                slideInHorizontally { fullWidth -> fullWidth } with
                        slideOutHorizontally { fullWidth -> -fullWidth }
            }
        ) { (prevWord, currentWord, nextWord) ->
            Row(
                modifier = Modifier
                    .animateContentSize()
                    .fillMaxSize()
                    .padding(horizontal = if (isFullScreen) 0.dp else 16.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                // Zone gauche : occupe la moitié de l'espace restant
                Box(modifier = Modifier.weight(1f)) {
                    Text(
                        text = prev,
                        color = Color.LightGray,
                        fontSize = 28.sp,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis,
                    )
                }

                Spacer(modifier = Modifier.width(12.dp))

                // Mot central : reste parfaitement centré
                AnimatedWord(current)


                Spacer(modifier = Modifier.width(12.dp))

                // Zone droite : occupe l’autre moitié de l'espace restant
                Box(modifier = Modifier.weight(1f)) {
                    Text(
                        text = next,
                        color = Color.LightGray,
                        fontSize = 28.sp,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis,
                    )
                }
            }
        }
    }
